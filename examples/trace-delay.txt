createOscillator id=src
Oscillator[src].type=sine
Oscillator[src].frequency.setValueAtTime(440, 0)
Oscillator[src].start(0)
createDelay id=delay
Delay[delay].delayTime.setValueAtTime(0.120, 0)
wrapBypass(dry=0, wet=1)
createEmitter id=emit -> connect(emitter.output, destination)
connect src -> delay
connect delay -> emit

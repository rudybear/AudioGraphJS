createOscillator id=osc
Oscillator[osc].type=sine
Oscillator[osc].frequency.setValueAtTime(330, 0)
Oscillator[osc].start(0)
createChannelSplitter id=split
createGain id=lGain
Gain[lGain].gain.setValueAtTime(0.950, 0)
createGain id=rGain
Gain[rGain].gain.setValueAtTime(0.316, 0)
createChannelMerger id=merge
connect osc -> split
connect split[out 0] -> lGain
connect split[out 1] -> rGain
connect lGain -> merge[in 0]
connect rGain -> merge[in 1]
connect output sink merge -> destination

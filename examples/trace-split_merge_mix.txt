createOscillator id=src
Oscillator[src].type=sine
Oscillator[src].frequency.setValueAtTime(220, 0)
Oscillator[src].start(0)
createChannelSplitter id=split
createBiquadFilter id=lp
Biquad[lp].type=lowpass
Biquad[lp].frequency.setValueAtTime(800, 0)
Biquad[lp].Q.setValueAtTime(0.707, 0)
wrapBypass(dry=0, wet=1)
createBiquadFilter id=hp
Biquad[hp].type=highpass
Biquad[hp].frequency.setValueAtTime(1200, 0)
Biquad[hp].Q.setValueAtTime(0.707, 0)
wrapBypass(dry=0, wet=1)
createChannelMerger id=merge
createAudioMixer id=mix
createGain id=out
Gain[out].gain.setValueAtTime(0.800, 0)
connect src -> split
connect split[out 0] -> lp
connect lp -> merge[in 0]
connect split[out 1] -> hp
connect hp -> merge[in 1]
connect merge -> mix
connect mix -> out
connect output sink out -> destination

createOscillator id=osc
Oscillator[osc].type=sine
Oscillator[osc].frequency.setValueAtTime(440, 0)
Oscillator[osc].start(0)
createChannelMixer id=mix
ChannelMixer[mix].channelCount=2
ChannelMixer[mix].channelInterpretation=speakers
createEmitter id=emit -> connect(emitter.output, destination)
connect osc -> mix
connect mix -> emit

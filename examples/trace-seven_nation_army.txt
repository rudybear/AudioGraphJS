createOscillator id=osc
Oscillator[osc].type=square
Oscillator[osc].frequency.setValueAtTime(82.410, 0)
Oscillator[osc].setPeriodicWave(pulseWidth=0.250)
Oscillator[osc].start(0)
createBiquadFilter id=lpf_pre
Biquad[lpf_pre].type=lowpass
Biquad[lpf_pre].frequency.setValueAtTime(800, 0)
Biquad[lpf_pre].Q.setValueAtTime(0.700, 0)
wrapBypass(dry=0, wet=1)
createWaveShaper id=dist
WaveShaper[dist].amount=0.700
WaveShaper[dist].oversample=2x
wrapBypass(dry=0, wet=1)
createBiquadFilter id=lpf_post
Biquad[lpf_post].type=lowpass
Biquad[lpf_post].frequency.setValueAtTime(1200, 0)
Biquad[lpf_post].Q.setValueAtTime(0.700, 0)
wrapBypass(dry=0, wet=1)
createGain id=amp
Gain[amp].gain.setValueAtTime(0, 0)
createDelay id=slap
Delay[slap].delayTime.setValueAtTime(0.070, 0)
wrapBypass(dry=0, wet=1)
createGain id=slapGain
Gain[slapGain].gain.setValueAtTime(0.150, 0)
connect osc -> lpf_pre
connect lpf_pre -> dist
connect dist -> lpf_post
connect lpf_post -> amp
connect amp -> slap
connect slap -> slapGain
connect output sink amp -> destination
connect output sink slapGain -> destination
